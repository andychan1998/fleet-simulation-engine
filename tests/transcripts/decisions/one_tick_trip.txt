# Decision: a trip's clock starts at the tick the command is issued. A 1-tick
# trip sent at tick T arrives at T+1, so the tick that commits it also
# completes it: the ship never shows "transit" in a committed state.

> {"command":"init","parameters":{"travel_times":[[0,1],[1,0]],"dock_throughput":[1,1],"vessels":[{"id":"v1","location":0}]}}

# planned state shows the trip
> {"command":"set_destination","parameters":{"vessel_id":"v1","destination":1}}
< {"status":"ok","payload":{"tick":0,"vessels":{"v1":{"state":"transit","from":0,"to":1,"arrives_at_tick":1,"cargo":[]}},"cargo":[]}}

# one tick later it has already arrived
> {"command":"tick","parameters":{}}
< {"status":"ok","payload":{"tick":1,"vessels":{"v1":{"state":"idle","location":1,"cargo":[]}},"cargo":[]}}
> {"command":"get_state_at","parameters":{"tick":0}}
< {"status":"ok","payload":{"tick":0,"vessels":{"v1":{"state":"idle","location":0,"cargo":[]}},"cargo":[]}}

# and it can be sent straight back
> {"command":"set_destination","parameters":{"vessel_id":"v1","destination":0}}
<~ {"payload":{"vessels":{"v1":{"state":"transit","from":1,"to":0,"arrives_at_tick":2}}}}
