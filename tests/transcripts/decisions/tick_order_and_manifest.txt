# Decision: tick runs in this order: pending commands take effect, the clock
# advances, ships arrive, the manifest's cargo lands, the state is saved.
# So cargo from a tick's manifest is part of that tick's state (and history)
# and can be loaded right after that tick.

> {"command":"init","parameters":{"travel_times":[[0,1],[1,0]],"dock_throughput":[2,2],"vessels":[{"id":"v1","location":0}]}}

# the dock commits and the cargo lands in the same tick
> {"command":"dock","parameters":{"vessel_id":"v1"}}
> {"command":"tick","parameters":{"cargo_manifest":[{"location":0,"destination":1,"quantity":1}]}}
< {"status":"ok","payload":{"tick":1,"vessels":{"v1":{"state":"docked","location":0,"cargo":[]}},"cargo":[{"location":0,"destination":1,"quantity":1}]}}
> {"command":"load","parameters":{"vessel_id":"v1","destination":1}}
<~ {"status":"ok"}
> {"command":"get_state_at","parameters":{"tick":1}}
< {"status":"ok","payload":{"tick":1,"vessels":{"v1":{"state":"docked","location":0,"cargo":[]}},"cargo":[{"location":0,"destination":1,"quantity":1}]}}

# a ship arrives on the same tick that new cargo lands at its port
> {"command":"undock","parameters":{"vessel_id":"v1"}}
> {"command":"tick","parameters":{}}
> {"command":"set_destination","parameters":{"vessel_id":"v1","destination":1}}
> {"command":"tick","parameters":{"cargo_manifest":[{"location":1,"destination":0,"quantity":1}]}}
< {"status":"ok","payload":{"tick":3,"vessels":{"v1":{"state":"idle","location":1,"cargo":[{"destination":1,"quantity":1}]}},"cargo":[{"location":1,"destination":0,"quantity":1}]}}
